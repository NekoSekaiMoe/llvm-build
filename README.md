# llvm-build
Build google llvm 20 in arm64 target

## Version
v1 command(recommend)

        cmake .. -DBUILD_SHARED_LIBS=OFF -DLLVM_BUILD_LLVM_DYLIB=OFF -DLLVM_ENABLE_PROJECTS="clang;lld;polly;compiler-rt" \
        -DCMAKE_BUILD_TYPE=Release -DLLVM_TARGETS_TO_BUILD="AArch64;ARM;X86" -DLLVM_CCACHE_BUILD=ON -DLLVM_ENABLE_Z3_SOLVER=OFF \
        -DLLVM_INCLUDE_BENCHMARKS=OFF -DLLVM_INCLUDE_EXAMPLES=OFF -DLLVM_INCLUDE_TESTS=OFF -DLLVM_INSTALL_UTILS=ON \
        -DLLVM_OPTIMIZED_TABLEGEN=ON -DLLVM_UNREACHABLE_OPTIMIZE=OFF -DLLVM_ENABLE_EH=ON -DLLVM_ENABLE_RTTI=ON \
        -DCMAKE_INTERPROCEDURAL_OPTIMIZATION=OFF -DLLVM_BUILD_RUNTIME=OFF -DLLVM_ENABLE_BINDINGS=OFF -DLLVM_ENABLE_LIBEDIT=OFF \
        -DLLVM_ENABLE_LIBPFM=OFF -DLLVM_ENABLE_LIBXML2=OFF -DLLVM_ENABLE_OCAMLDOC=OFF -DLLVM_ENABLE_PLUGINS=OFF -DCLANG_INCLUDE_DOCS=OFF \
        -DCLANG_INCLUDE_TESTS=OFF -DCLANG_TOOL_CLANG_IMPORT_TEST_BUILD=OFF -DCLANG_TOOL_CLANG_LINKER_WRAPPER_BUILD=OFF \
        -DCLANG_TOOL_C_INDEX_TEST_BUILD=OFF -DLLD_INCLUDE_TESTS=OFF -DLLD_INCLUDE_DOCS=OFF -DPOLLY_INCLUDE_TESTS=OFF -DPOLLY_INCLUDE_DOCS=OFF \
        -GNinja
        DESTDIR=/home/runner/work/llvm-build/llvm-build/out ninja install/strip
